@module(entry="static_online_attend", target=CudaTarget("nvidia.h200_sxm"), topologies=(Topology("cta", 132),))
class static_online_attend:
    @func
    def static_online_attend(
        q: Tensor[(1, 1, 32, 128), "bf16"],
        k_cache: Tensor[(1, 4096, 4, 128), "bf16"],
        v_cache: Tensor[(1, 4096, 4, 128), "bf16"]
    ) -> Tensor[(1, 1, 32, 128), "bf16",
        ShardLayout(
            layout=Layout((1, 1, 32, 128), (4096, 4096, 128, 1)),
            attrs=(B(),),
            mesh=Mesh((Topology("cta", 132),), Layout((132,), (1,))),
        )]:
        with cta as _cta:
            q_sh = reshard(q, layout=ShardLayout(
                layout=Layout((1, 1, 32, 128), (4096, 4096, 128, 1)),
                attrs=(B(),),
                mesh=Mesh((Topology("cta", 132),), Layout((132,), (1,))),
            ))
            q_f = cast(q_sh, dtype="f32")
            tmpl = reduce(q_f, axes=(-1,), keepdim=True, kind="sum")
            l = full_like(tmpl, value=0.0)
            o = full_like(q_f, value=0.0)
            m = full_like(tmpl, value=-1e+30)
            for i in range(4096):
                v0 = full_like(q_f, value=0.08838834764831843)
                q_s = mul(q_f, v0)
                v1 = reshape(i, new_shape=(1,))
                v2 = index_select(k_cache, v1, dim=1)
                v3 = reshape(v2, new_shape=(1, 4, 128))
                v4 = repeat_interleave(v3, repeats=8, axis=1)
                v5 = cast(v4, dtype="f32")
                k_i = reshape(v5, new_shape=(1, 1, 32, 128))
                v6 = mul(q_s, k_i)
                score = reduce(v6, axes=(-1,), keepdim=True, kind="sum")
                m_new = max(m, score)
                v7 = sub(m, m_new)
                corr = exp(v7)
                v8 = mul(l, corr)
                v9 = sub(score, m_new)
                p = exp(v9)
                l_2 = add(v8, p)
                v10 = mul(o, corr)
                v11 = reshape(i, new_shape=(1,))
                v12 = index_select(v_cache, v11, dim=1)
                v13 = reshape(v12, new_shape=(1, 4, 128))
                v14 = repeat_interleave(v13, repeats=8, axis=1)
                v15 = cast(v14, dtype="f32")
                v_i = reshape(v15, new_shape=(1, 1, 32, 128))
                v16 = mul(p, v_i)
                o_2 = add(v10, v16)
                l = l_2
                o = o_2
                m = m_new
            v18 = div(o, l)
            v19 = cast(v18, dtype="bf16")
            return v19

from tilefoundry.ir.types.shard import (
    B, S, P, ComposedLayout, Layout, Mesh, ShardLayout, Topology,
)
from tilefoundry.ir.types.dim import DimVar, ceildiv

seq_len = DimVar("seq_len", 0, 8193)

cta = Mesh((Topology("cta", 128),), Layout((32, 4), (4, 1)), names=('strip', 'tile'))

@module(entry="project", target=CudaTarget("nvidia.h200_sxm"), topologies=(Topology("cta", 128),))
class _SymbolicStoreOffset:
    @func
    def project(
        x: Tensor[(16, seq_len, 128), "bf16"],
        weight: ConstTensor[(96, 2048, 128), "bf16"],
        out: Tensor[(96, seq_len, 128), "bf16"]
    ) -> Tensor[(96, seq_len, 128), "bf16"]:
        with cta as _cta:
            for position in range(0, seq_len, 256):
                for group in range(3):
                    v1 = 0
                    base = add(position, v1)
                    v4 = slice(x, (0, base, 0), sizes=(1, 256, 128), strides=(1, 1, 1))
                    rows = reshard(v4, layout=ShardLayout(
                        layout=Layout((1, 4, 64, 128), None),
                        attrs=(B(), S(1)),
                        mesh=cta,
                    ), storage=smem)
                    v9 = weight[0:32, 0:128, :]
                    weights = reshard(v9, layout=ShardLayout(
                        layout=Layout((32, 128, 128), None),
                        attrs=(S(0), B()),
                        mesh=cta,
                    ), storage=smem)
                    product = matmul(rows, weights, a_layout="MK", b_layout="KN")
                    for head in range(1, 16, 1):
                        v10 = 0
                        head_index = add(head, v10)
                        v13 = slice(x, (head_index, base, 0), sizes=(1, 256, 128), strides=(1, 1, 1))
                        next_rows = reshard(v13, layout=ShardLayout(
                            layout=Layout((1, 4, 64, 128), None),
                            attrs=(B(), S(1)),
                            mesh=cta,
                        ), storage=smem)
                        v15 = 128
                        weight_base = mul(head_index, v15)
                        v18 = slice(weight, (0, weight_base, 0), sizes=(32, 128, 128), strides=(1, 1, 1))
                        next_weights = reshard(v18, layout=ShardLayout(
                            layout=Layout((32, 128, 128), None),
                            attrs=(S(0), B()),
                            mesh=cta,
                        ), storage=smem)
                        v19 = matmul(next_rows, next_weights, a_layout="MK", b_layout="KN")
                        product_2 = add(product, v19)
                        product = product_2
                    v21 = 0
                    group_index = add(group, v21)
                    v22 = 32
                    store_base = mul(group_index, v22)
                    result_2 = insert_slice(result, product, (store_base, base, 0))
                    result = result_2
                result = result
            return result

class Weighted:
    @func
    def scaled(
        x: Tensor[(168,), "f32"],
        w: ConstTensor[(168,), "f32"]
    ) -> Tensor[(168,), "f32", (168 @ cta.block,)]:

            v0 = reshard(weighted, layout=ShardLayout(
                layout=Layout((168,), None),
                attrs=(S(0),),
                mesh=cta,
            ), storage=gmem)
