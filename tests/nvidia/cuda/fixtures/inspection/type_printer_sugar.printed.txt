from __future__ import annotations

from tilefoundry.dsl import *
from tilefoundry_nvidia import cuda

@module(entry="composed_mesh_pipeline", target=cuda.Target("nvidia.h200_sxm"), topologies=(Topology("cta", 4), Topology("thread", 8),))
class TypePrinterSugar:
    thread = Mesh(("thread",), layout=(2, 4), names=("warp", "lane"))
    cta = Mesh(("cta",), layout=(4,), names=("tile",))
    cta_2 = Mesh((Topology("cta", 4), Topology("thread", 8)), layout=(4, 2, 4), names=("tile", "warp", "lane"))

    @func
    def nested_loop_tuple(
        x: Tensor[(8, 16), "f32"],
        weight: Tensor[(8, 16), "f32", (8, 16) @ {thread.lane @ P("max")}]
    ):
        with Mesh(("thread",), layout=(8,), names=("lane",)) as thread_2:
            split = tf.reshard(x, layout=(8 @ thread_2.lane, 16), storage="rmem")
            whole = tf.reshard(x, layout=(8, 16), storage="rmem")
            for _ in range(3):
                whole_2 = tf.add(whole, whole)
                split_2 = tf.square(split)
                split = split_2
                whole = whole_2
            v7 = tf.reshard(split, layout=(8, 16), storage="gmem")
            v10 = tf.reshard(whole, layout=(8, 16), storage="gmem")
            with thread:
                per_warp = tf.reshard(weight, layout=(2 @ thread.warp, 4, 16), storage="rmem")
                unfolded = tf.reshard(per_warp, layout=(8, 16), storage="gmem")
            return (v7, v10, unfolded)

    @func
    def named_and_out_of_scope(
        x: Tensor[(8, 16), "f32"],
        held: Tensor[(8, 16), "f32", (4 @ cta.tile, 2, 16), "rmem"],
        frag: Tensor[(16,), "f32", ((2 @ thread.warp, 4 @ thread.lane, 2), (8, 2, 1)), "rmem"]
    ):
        with cta:
            mine = tf.reshard(x, layout=(4 @ cta.tile, 2, 16), storage="rmem")
            v1 = tf.reshard(mine, layout=(8, 16), storage="gmem")
            escaped = tf.reshard(x, layout=ShardLayout(
                layout=Layout((2, 4, 16), (0, 16, 1)),
                attrs=(S(0), B()),
                mesh=thread,
            ), storage="rmem")
            return (v1, held, frag, escaped)

    @func
    def composed_mesh_pipeline(
        x: Tensor[(8, 4, 16), "f32"],
        seed: Tensor[(16,), "f32"],
        acc: Tensor[(8, 16), "f32", (2 @ thread.warp, 4, 16) @ {thread.lane @ P("sum")}, "rmem"],
        mixed: Tensor[(8, 16), "f32", (4 @ cta_2.tile, 2, 16) @ {cta_2.lane @ P("sum")}, "rmem"]
    ):
        with thread:
            with cta:
                composed = tf.reshard(x, layout=(4 @ cta.tile, 2, 2 @ thread.warp, 2, 4 @ thread.lane, 4), storage="rmem")
                v1 = tf.square(composed)
                staged = tf.reshard(v1, layout=(4 @ cta.tile, 2, 4, 16), storage="smem")
                narrowed = tf.cast(staged, dtype="bf16")
                swapped = tf.transpose(narrowed, perm=(0, 2, 1))
                gathered = tf.reshard(swapped, layout=(8, 16, 4), storage="gmem")
                seeded = tf.reshard(seed, layout=((2 @ thread.warp, 4 @ thread.lane, 2), (8, 2, 1)), storage="rmem")
            folded = tf.reshard(acc, layout=(2 @ thread.warp, 4, 4 @ thread.lane, 4), storage="rmem")
            summed = tf.reshard(mixed, layout=(8, 16), storage="rmem")
            for _ in range(3):
                summed_2 = tf.add(summed, summed)
                folded_2 = tf.square(folded)
                folded = folded_2
                summed = summed_2
            v18 = tf.reshard(folded, layout=(8, 16), storage="gmem")
            v21 = tf.reshard(summed, layout=(8, 16), storage="gmem")
            v24 = tf.reshard(seeded, layout=(16,), storage="gmem")
            return (gathered, v18, v21, v24)
